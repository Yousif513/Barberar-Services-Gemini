   44: // An authenticator (TOTP) verification that failed. attemptsLeft comes from admin_session_state after the failure.
   45: export function totpErrorMessage(error: unknown, locale: AuthLocale, attemptsLeft: number | null = null): string {
   46:   const t = copy[locale];
   47:   const { code, status, message, name } = fields(error);
   48:   if (/locked/i.test(message) || attemptsLeft === 0) return locale === "ar" ? t.locked.replace("{n}", arabicDigits(10)) : t.locked;
   49:   if (code === "mfa_verification_failed" || /invalid totp code/i.test(message) || status === 422) {
   50:     const base = locale === "ar" ? t.wrongTotp.replace("{s}", arabicDigits(30)) : t.wrongTotp;
   51:     if (attemptsLeft === null || attemptsLeft > 5) return base;
   52:     const left = attemptsLeft === 1 ? t.lastAttempt : t.attemptsLeft.replace("{n}", locale === "ar" ? arabicDigits(attemptsLeft) : String(attemptsLeft));
   53:     return `${base} ${left}`;
   54:   }
   55:   if (code === "mfa_challenge_expired") return t.challengeExpired;
   56:   if (code.startsWith("over_") || status === 429) return t.rateLimited;
   57:   if (name === "AuthRetryableFetchError" || /failed to fetch|network/i.test(message)) return t.network;
   58:   return t.generic;
